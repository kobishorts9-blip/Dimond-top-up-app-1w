            // --- NOTIFICATION AND BAN SYSTEM FUNCTIONS ---
            async function loadNotificationsAndBanStatus() {
                if (!currentUser) return;
                
                try {
                    // নোটিফিকেশন লোড
                    const notificationsRef = db.collection('notifications')
                        .where('targetUsers', 'array-contains', currentUser.uid)
                        .where('status', '==', 'active')
                        .orderBy('createdAt', 'desc')
                        .limit(5);
                    
                    const notificationsSnapshot = await notificationsRef.get();
                    
                    if (!notificationsSnapshot.empty) {
                        const notificationAlert = document.getElementById('notification-alert');
                        const notificationMessage = document.getElementById('notification-message');
                        
                        notificationsSnapshot.forEach(doc => {
                            const notification = doc.data();
                            const now = new Date();
                            const expiresAt = notification.expiresAt?.toDate();
                            
                            if (!expiresAt || expiresAt > now) {
                                notificationAlert.classList.remove('d-none');
                                notificationMessage.textContent = notification.message;
                                
                                if (notification.type === 'warning') {
                                    notificationAlert.style.background = 'var(--warning-bg)';
                                    notificationAlert.style.borderLeftColor = 'var(--warning-color)';
                                } else if (notification.type === 'info') {
                                    notificationAlert.style.background = 'var(--info-bg)';
                                    notificationAlert.style.borderLeftColor = '#0c5460';
                                } else if (notification.type === 'danger') {
                                    notificationAlert.style.background = 'var(--danger-bg)';
                                    notificationAlert.style.borderLeftColor = 'var(--danger-color)';
                                }
                                
                                if (notification.autoClose) {
                                    setTimeout(() => {
                                        notificationAlert.classList.add('d-none');
                                    }, notification.autoCloseSeconds * 1000);
                                }
                            }
                        });
                    }
                    
                    // ব্যান স্ট্যাটাস চেক
                    const userDoc = await db.collection('users').doc(currentUser.uid).get();
                    if (userDoc.exists) {
                        const userData = userDoc.data();
                        
                        if (userData.isBanned) {
                            isUserBanned = true;
                            showBanStatus(userData);
                            
                            if (userData.banUntil) {
                                const banDate = userData.banUntil.toDate();
                                if (banDate < new Date()) {
                                    await db.collection('users').doc(currentUser.uid).update({
                                        isBanned: false,
                                        banReason: '',
                                        banUntil: null
                                    });
                                    isUserBanned = false;
                                    hideBanStatus();
                                }
                            }
                        } else {
                            isUserBanned = false;
                            hideBanStatus();
                        }
                    }
                    
                } catch (error) {
                    console.error("Error loading notifications/ban status: ", error);
                }
            }

            // ব্যান স্ট্যাটাস দেখানো
            function showBanStatus(userData) {
                const banAlert = document.getElementById('ban-status-alert');
                const banReason = document.getElementById('ban-reason');
                const banUntil = document.getElementById('ban-until');
                
                const banDepositAlert = document.getElementById('ban-status-deposit');
                const banReasonDeposit = document.getElementById('ban-reason-deposit');
                const banUntilDeposit = document.getElementById('ban-until-deposit');
                
                if (banAlert) {
                    banAlert.classList.remove('d-none');
                    banReason.textContent = userData.banReason || 'Violation of terms of service';
                    
                    if (userData.banUntil) {
                        const banDate = userData.banUntil.toDate();
                        banUntil.textContent = banDate.toLocaleDateString() + ' ' + banDate.toLocaleTimeString();
                    } else {
                        banUntil.textContent = 'Permanent';
                    }
                }
                
                if (banDepositAlert) {
                    banDepositAlert.classList.remove('d-none');
                    banReasonDeposit.textContent = userData.banReason || 'Violation of terms of service';
                    
                    if (userData.banUntil) {
                        const banDate = userData.banUntil.toDate();
                        banUntilDeposit.textContent = banDate.toLocaleDateString() + ' ' + banDate.toLocaleTimeString();
                    } else {
                        banUntilDeposit.textContent = 'Permanent';
                    }
                }
                
                if (isUserBanned) {
                    disableUserActions();
                }
            }

                    .where('targetUsers', 'array-contains', currentUser.uid)
                    .where('status', '==', 'active')
                    .onSnapshot((snapshot) => {
                        snapshot.docChanges().forEach((change) => {
                            if (change.type === 'added') {
                                const notification = change.doc.data();
                                showNewNotification(notification);
                            }
                        });
                    });
                
                db.collection('users').doc(currentUser.uid)
                    .onSnapshot((doc) => {
                        if (doc.exists) {

                            if (userData.isBanned) {
                                isUserBanned = true;
                                showBanStatus(userData);
                            } else {
                                isUserBanned = false;
                                hideBanStatus();
                            }
                        }
                    });
            }

            // থিম অ্যাপ্লাই ফাংশন
            function applyTheme(color) {
                const root = document.documentElement;
                root.style.setProperty('--primary-color', color);
                
                const gradient = `linear-gradient(45deg, ${color}, ${darkenColor(color, 20)})`;
                root.style.setProperty('--primary-gradient', gradient);
                
                const glow = `0 0 20px ${hexToRGBA(color, 0.6)}`;
                root.style.setProperty('--primary-glow', glow);
            }
            
            // কালার হেল্পার ফাংশন
            function darkenColor(color, percent) {
                const num = parseInt(color.replace("#", ""), 16);
                const amt = Math.round(2.55 * percent);
                const R = (num >> 16) - amt;
                const G = (num >> 8 & 0x00FF) - amt;
                const B = (num & 0x0000FF) - amt;
                return "#" + (0x1000000 + (R < 255 ? R < 1 ? 0 : R : 255) * 0x10000 +
                              (G < 255 ? G < 1 ? 0 : G : 255) * 0x100 +
                              (B < 255 ? B < 1 ? 0 : B : 255)).toString(16).slice(1);
            }
            
            function hexToRGBA(hex, alpha) {
                const r = parseInt(hex.slice(1, 3), 16);
                const g = parseInt(hex.slice(3, 5), 16);
                const b = parseInt(hex.slice(5, 7), 16);
                return `rgba(${r}, ${g}, ${b}, ${alpha})`;
            }

                    const method = e.target.dataset.method;

                    const minDepositAmount = globalSettings.minDepositAmount || 0;

                        hideLoader();
                        return;
                    }

                    const depositData = {
                        uid: currentUser.uid,
                        userEmail: currentUser.email,
                        amount: amount,
                        method: method,
                        transactionId: transactionId,
                        status: 'Pending',
                        timestamp: firebase.firestore.FieldValue.serverTimestamp()
                    };
                    
                    try {
                        const topLevelDocRef = db.collection('deposits').doc();
                        const userSubDocRef = db.collection('users').doc(currentUser.uid).collection('deposits').doc(topLevelDocRef.id);
                        const batch = db.batch();
                        batch.set(topLevelDocRef, depositData);
                        batch.set(userSubDocRef, depositData);
                        await batch.commit();
                        showAlert('Deposit request submitted successfully!', 'success');
                        form.reset();
                        switchAppPage('account-page');
                    } catch (error) {
                        console.error("Error submitting deposit: ", error);
                        showAlert('Failed to submit deposit request. Please try again.', 'error');
                    } finally {
                        hideLoader();
                    }

            // Place Order
            document.getElementById('order-form').addEventListener('submit', (e) => {
                e.preventDefault();
                
                if (isUserBanned) {
                    showAlert('Your account is banned. Cannot place orders.', 'error');
                    orderModal.classList.remove('active');
                    return;
                }
                
                showLoader();
                const playerID = document.getElementById('player-id').value;
                if (!playerID.trim()) {
                    showAlert('Please enter a Player ID.', 'error');
                    hideLoader();
                    return;
                }
                
                const userRef = db.collection('users').doc(currentUser.uid);
                
                db.runTransaction(async (transaction) => {
                    const userDoc = await transaction.get(userRef);
                    if (!userDoc.exists) {
                        throw "User document does not exist!";
                    }

                    const currentBalance = userDoc.data().walletBalance;
                    
                    if (currentBalance <= 0) {
                        throw "ZERO_BALANCE";
                    }
                    if (currentBalance < currentProduct.price) {
                        throw "INSUFFICIENT_BALANCE";
                    }

                    const newBalance = currentBalance - currentProduct.price;
                    transaction.update(userRef, { walletBalance: newBalance });
                    
                    const orderData = {
                        uid: currentUser.uid,
                        userEmail: currentUser.email,
                        playerID: playerID,
                        productName: currentProduct.name,
                        productId: currentProduct.id,
                        price: currentProduct.price,
                        status: 'Pending',
                        timestamp: firebase.firestore.FieldValue.serverTimestamp()
                    };
                    const newOrderRef = db.collection('orders').doc();
                    transaction.set(newOrderRef, orderData);
                    transaction.set(userRef.collection('orders').doc(newOrderRef.id), orderData);
                    
                }).then(() => {
                    showAlert('Order placed successfully!', 'success');
                    orderModal.classList.remove('active');
                    document.getElementById('order-form').reset();
                }).catch(error => {
                    console.error("Order transaction failed: ", error);
                    let errorMessage = `Order failed: ${error}`;
                    if (error === "ZERO_BALANCE") {
                        errorMessage = 'আপনার ওয়ালেটে কোন টাকা নেই। দয়া করে টাকা যোগ করুন। (Your wallet has no money. Please add money.)';
                    } else if (error === "INSUFFICIENT_BALANCE") {
                         errorMessage = 'Insufficient balance to place this order.';
                    }
                    showAlert(errorMessage, 'error');
                }).finally(() => {
                    hideLoader();
                });
            });

            document.getElementById('new-password').addEventListener('input', function(e) {
                const password = e.target.value;
                const strengthBar = document.getElementById('password-strength-bar');
                
                let strength = 0;
                
                if (password.length >= 6) strength += 25;
                if (password.length >= 8) strength += 25;
                
                if (/[A-Z]/.test(password)) strength += 25;
                if (/[0-9]/.test(password)) strength += 25;
                if (/[^A-Za-z0-9]/.test(password)) strength += 25;
                
                strengthBar.style.width = Math.min(strength, 100) + '%';
                
                if (strength < 50) {
                    strengthBar.className = 'strength-bar strength-weak';
                } else if (strength < 75) {
                    strengthBar.className = 'strength-bar strength-medium';
                } else {
                    strengthBar.className = 'strength-bar strength-strong';
                }
            });

                    const credential = firebase.auth.EmailAuthProvider.credential(
                        user.email, 
                        currentPassword
                    );
                    
                    await user.reauthenticateWithCredential(credential);

                const slider = document.querySelector('.slider');
                if (!slider || slider.children.length <= 1) return;
                let currentIndex = 0;
                const slides = document.querySelectorAll('.slide');
                const totalSlides = slides.length;
                setInterval(() => {
                    currentIndex = (currentIndex + 1) % totalSlides;
                    slider.style.transform = `translateX(-${currentIndex * 100}%)`;
                }, 3000);
            }

                }
                @keyframes slideOut {
